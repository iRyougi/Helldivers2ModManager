## Context: How the Codebase Works

- **Mod storage:** `{StorageDir}/Mods/{ModName}/` — each mod dir has `manifest.json` + patch files
- **Patch files:** Named `[16-hex-chars].patch_N` (and `.patch_N.gpu_resources`, `.patch_N.stream`)
- **Conflict = same 16-char prefix** in two different mods' deployable files
- **Deploy process:** Scans files, groups by 16-char prefix, assigns sequential patch indices
- **MessageBox system:** Uses `WeakReferenceMessenger` — send `MessageBoxConfirmMessage` for yes/no, `MessageBoxWarningMessage` for info-only
- **Localization:** All strings in `LocalizationService.CreateDefaultLanguageFilesAsync()` — add to BOTH `enTranslations` and `zhCnTranslations` dictionaries; bump `CurrentTranslationVersion`
- **ModViewModel.Order:** Does not exist yet — needs adding

```csharp
public ConflictInfo[] DetectConflicts(Guid[] enabledModGuids)
{
    GuardInitialized();

    // Map: 16-char prefix → list of mods that have files with that prefix
    var prefixToMods = new Dictionary<string, List<ModData>>();

    foreach (var guid in enabledModGuids)
    {
        var mod = GetModByGuid(guid);
        if (mod is null) continue;

        var prefixes = new HashSet<string>();

        void ScanDir(DirectoryInfo dir)
        {
            if (!dir.Exists) return;
            foreach (var file in dir.GetFiles())
            {
                if (!GetPatchFileRegex().IsMatch(file.Name)) continue;
                var prefix = file.Name[0..16];
                prefixes.Add(prefix);
            }
        }

        switch (mod.Manifest.Version)
        {
            case ManifestVersion.Legacy:
            {
                var man = (LegacyModManifest)mod.Manifest;
                if (man.Options is { } opts && mod.SelectedOptions is [var sel, ..])
                    ScanDir(new DirectoryInfo(Path.Combine(mod.Directory.FullName, opts[sel])));
                else
                    ScanDir(mod.Directory);
                break;
            }
            case ManifestVersion.V1:
            {
                var man = (V1ModManifest)mod.Manifest;
                if (man.Options is null)
                {
                    ScanDir(mod.Directory);
                    break;
                }
                var enabled = mod.EnabledOptions;
                var selected = mod.SelectedOptions;
                for (int i = 0; i < man.Options.Count; i++)
                {
                    if (i >= enabled.Length || !enabled[i]) continue;
                    var opt = man.Options[i];
                    if (opt.Include is { } incs)
                        foreach (var inc in incs)
                            ScanDir(new DirectoryInfo(Path.Combine(mod.Directory.FullName, inc)));
                    if (opt.SubOptions is { } subs && i < selected.Length)
                        foreach (var inc in subs[selected[i]].Include)
                            ScanDir(new DirectoryInfo(Path.Combine(mod.Directory.FullName, inc)));
                }
                break;
            }
        }

        foreach (var prefix in prefixes)
        {
            if (!prefixToMods.ContainsKey(prefix))
                prefixToMods[prefix] = new List<ModData>();
            prefixToMods[prefix].Add(mod);
        }
    }

    return prefixToMods
        .Where(kvp => kvp.Value.Count > 1)
        .Select(kvp => new ConflictInfo
        {
            PatchPrefix = kvp.Key,
            ConflictingMods = kvp.Value.ToArray()
        })
        .ToArray();
}
```

```csharp
public async Task<IModManifest?> PeekManifestFromArchiveAsync(FileInfo archiveFile)
{
    var tmpDir = new DirectoryInfo(Path.Combine(
        _settingsService!.TempDirectory,
        "peek_" + archiveFile.Name[..^archiveFile.Extension.Length]));
    try
    {
        if (tmpDir.Exists) tmpDir.Delete(true);
        tmpDir.Create();
        await Task.Run(() => SharpCompress.Archives.ArchiveFactory.Open(archiveFile)
            .ExtractToDirectory(tmpDir.FullName));
        var manifestFile = new FileInfo(Path.Combine(tmpDir.FullName, "manifest.json"));
        if (!manifestFile.Exists) return null;
        return ModManifest.DeserializeFromFile(manifestFile);
    }
    catch
    {
        return null;
    }
    finally
    {
        if (tmpDir.Exists) tmpDir.Delete(true);
    }
}
```

```csharp
protected override async void OnStartup(StartupEventArgs e)
{
    // Single instance enforcement
    _singleInstanceMutex = new Mutex(true, MutexName, out bool isNewInstance);
    if (!isNewInstance)
    {
        // Another instance is running — bring it to front
        var current = System.Diagnostics.Process.GetCurrentProcess();
        var others = System.Diagnostics.Process.GetProcessesByName(current.ProcessName)
            .Where(p => p.Id != current.Id)
            .ToArray();
        foreach (var other in others)
        {
            var hwnd = other.MainWindowHandle;
            if (hwnd != nint.Zero)
            {
                NativeMethods.ShowWindow(hwnd, NativeMethods.SW_RESTORE);
                NativeMethods.SetForegroundWindow(hwnd);
            }
        }
        Shutdown();
        return;
    }

    base.OnStartup(e);
    // ... rest of existing OnStartup code
```
